# Очередь повторений

> [!tip] Как пользоваться
> Только эти две кнопки. «Повторил» — интервал × 2.5. «Провал» — интервал 1 день, статус «растёт».
> Больше 10 минут не тратить. Не помнишь — честно жми «Провал».

## Что просрочено

```dataviewjs
const today = dv.luxon.DateTime.now().startOf("day");

function toDay(raw) {
  if (!raw) return null;
  if (dv.luxon.DateTime.isDateTime(raw)) return raw.startOf("day");
  const d = dv.luxon.DateTime.fromISO(String(raw));
  return d.isValid ? d.startOf("day") : null;
}

const pages = dv.pages('"03_Ресурсы/Концепты"')
  .where(p => p.type === "концепт" && p.review === true && p.review_due);

const rows = [];
for (const p of pages) {
  const due = toDay(p.review_due);
  if (!due) continue;
  const delta = Math.round(due.diff(today, "days").days);
  if (delta <= 0) {
    rows.push([
      p.file.link,
      p.feynman_score ?? 0,
      p.review_interval ?? 1,
      delta === 0 ? "сегодня" : `${-delta} дн. назад`
    ]);
  }
}

rows.sort((a, b) => Number(b[1]) - Number(a[1]));

if (rows.length === 0) {
  dv.paragraph("Очередь пуста. Возвращайся завтра.");
} else {
  dv.table(["Концепт", "Балл", "Интервал", "Просрочено"], rows);
}
```

## Оценка

```dataviewjs
const today = dv.luxon.DateTime.now().startOf("day");

function toDay(raw) {
  if (!raw) return null;
  if (dv.luxon.DateTime.isDateTime(raw)) return raw.startOf("day");
  const d = dv.luxon.DateTime.fromISO(String(raw));
  return d.isValid ? d.startOf("day") : null;
}

const pages = dv.pages('"03_Ресурсы/Концепты"')
  .where(p => p.type === "концепт" && p.review === true && p.review_due);

const due = [];
for (const p of pages) {
  const d = toDay(p.review_due);
  if (d && d <= today) due.push(p);
}

async function rate(page, passed) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  if (!file) return;
  await app.fileManager.processFrontMatter(file, (fm) => {
    let interval = Number(fm.review_interval ?? 1);
    let score = Number(fm.feynman_score ?? 0);
    if (passed) {
      interval = Math.min(180, Math.max(2, Math.round(interval * 2.5)));
      score = Math.min(3, score + 1);
    } else {
      interval = 1;
      score = Math.max(0, score - 1);
      fm.status = "растёт";
    }
    fm.review_interval = interval;
    fm.feynman_score = score;
    fm.review_due = dv.luxon.DateTime.now().plus({ days: interval }).toISODate();
  });
  app.commands.executeCommandById("dataview:dataview-force-refresh-views");
}

if (due.length === 0) {
  dv.paragraph("Сейчас повторять нечего.");
} else {
  for (const p of due) {
    const row = dv.el("div", "", { cls: "review-row" });
    row.style.cssText = "display:flex;align-items:center;gap:8px;margin:6px 0;flex-wrap:wrap;";

    const link = dv.fileLink(p.file.path, false, p.file.name);
    link.style.flex = "1 1 auto";
    row.appendChild(link);

    const ok = dv.el("button", "Повторил", { container: row });
    ok.style.cssText = "padding:2px 10px;border-radius:6px;cursor:pointer;";

    const fail = dv.el("button", "Провал", { container: row });
    fail.style.cssText = "padding:2px 10px;border-radius:6px;cursor:pointer;";

    ok.onclick = () => rate(p, true);
    fail.onclick = () => rate(p, false);
  }
}
```

## Все концепты на повторении

```dataview
TABLE feynman_score AS "Балл", review_interval AS "Интервал", review_due AS "Следующий", status AS "Статус"
FROM "03_Ресурсы/Концепты"
WHERE type = "концепт" AND review = true
SORT review_due ASC
```
