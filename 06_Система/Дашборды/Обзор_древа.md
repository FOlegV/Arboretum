# Обзор древа

> [!info] Гейты
> Уровни 1–3 считаются автоматически по данным. Уровни 4–5 присваиваются вручную.

```dataviewjs
const skills = dv.pages('"02_Области/Навыки"')
  .where(p => p.type === "навык")
  .sort(p => p.file.name);

const concepts = dv.pages('"03_Ресурсы/Концепты"').where(p => p.type === "концепт");
const sessions = dv.pages('"Learn"').where(p => p.type === "сессия_фейнмана");
const projects = dv.pages('"01_Проекты"').where(p => p.status === "завершён");

function linked(pages, skill) {
  return pages.where(p => p.skill && p.skill.path === skill.file.path);
}

const rows = [];
for (const s of skills) {
  const c = linked(concepts, s);
  const sess = linked(sessions, s);
  const done = sess.where(p => p.result === "упрощён");
  const proj = linked(projects, s);

  const cCount = c.length;
  const sCount = sess.length;
  const dCount = done.length;
  const pCount = proj.length;

  let autoLevel = 0;
  if (cCount >= 1) autoLevel = 1;
  if (autoLevel === 1 && dCount >= 1) autoLevel = 2;
  if (autoLevel === 2 && pCount >= 1) autoLevel = 3;

  const stored = Number(s.level ?? 0);
  const level = Math.max(autoLevel, stored);

  let gate;
  if (cCount < 1) gate = "нужен ≥1 концепт";
  else if (dCount < 1) gate = "нужна ≥1 сессия «упрощён»";
  else if (pCount < 1) gate = "нужен ≥1 завершённый проект";
  else if (sCount < 3 || pCount < 2) gate = `сессий ${sCount}/3, проектов ${pCount}/2`;
  else gate = "уровень 5 — вручную";

  rows.push([s.file.link, level, cCount, dCount, pCount, gate]);
}

if (rows.length === 0) {
  dv.paragraph("Пока нет навыков. Создай первый в `02_Области/Навыки`.");
} else {
  dv.table(["Навык", "Уровень", "Концепты", "Сессии", "Проекты", "Следующий гейт"], rows);
}
```
